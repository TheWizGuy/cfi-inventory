<script>
const API_URL = "https://script.google.com/macros/s/AKfycbzB84656xhlhkEVRBsgpgrMxh4Iijty9Y7TkJUOHxx1EVrZdyxCn4Y60Lat09HyMUbW5w/exec";
let inventory = [];

// LOAD INVENTORY
fetch(API_URL)
  .then(res => res.json())
  .then(data => {
    inventory = data;

    document.getElementById("loading").innerText = "Inventory Loaded ✅";
    document.getElementById("addBtn").disabled = false;

    addItem();
  })
  .catch(err => {
    document.getElementById("loading").innerText = "Failed to load inventory ❌";
    console.error(err);
  });

// PURPOSE LOGIC
function handlePurpose() {
  const val = document.getElementById("purpose").value;
  const otherField = document.getElementById("otherPurposeField");

  if (val === "Other") {
    otherField.classList.remove("hidden");
  } else {
    otherField.classList.add("hidden");
  }
}

// ADD ITEM
function addItem() {

  if (inventory.length === 0) {
    alert("Inventory not loaded yet!");
    return;
  }

  const div = document.createElement("div");
  div.className = "item-row";

  const category = document.createElement("select");
  const item = document.createElement("select");
  const qty = document.createElement("input");
  const del = document.createElement("button");

  qty.type = "number";
  qty.placeholder = "Qty";
  qty.min = "1";

  del.innerText = "X";
  del.className = "btn btn-del";
  del.onclick = () => div.remove();

  const cats = [...new Set(inventory.map(i => i.category))];

  cats.forEach(c => {
    const opt = document.createElement("option");
    opt.value = c;
    opt.innerText = c;
    category.appendChild(opt);
  });

  category.onchange = () => {
    item.innerHTML = "";

    inventory
      .filter(i => i.category === category.value)
      .forEach(i => {

        const available = (i.total || 0) - (i.inUse || 0); // ✅ UPDATED LOGIC

        const opt = document.createElement("option");
        opt.value = i.item;
        opt.innerText = `${i.item} (Available: ${available})`;

        item.appendChild(opt);
      });
  };

  category.onchange();

  div.appendChild(category);
  div.appendChild(item);
  div.appendChild(qty);
  div.appendChild(del);

  document.getElementById("items").appendChild(div);
}

// SUBMIT
function submitForm() {

  let purpose = document.getElementById("purpose").value;

  if (purpose === "Other") {
    purpose = document.getElementById("otherPurpose").value;
  }

  const items = [];

  document.querySelectorAll(".item-row").forEach(row => {
    const selects = row.querySelectorAll("select");
    const qty = row.querySelector("input").value;

    items.push({
      category: selects[0].value,
      item: selects[1].value,
      qty: Number(qty)
    });
  });

  const payload = {
    name: document.getElementById("name").value,
    year: document.getElementById("year").value,
    dept: document.getElementById("dept").value,
    roll: document.getElementById("roll").value,
    contact: document.getElementById("contact").value,
    email: document.getElementById("email").value,
    borrowDate: document.getElementById("borrowDate").value,
    returnDate: document.getElementById("returnDate").value,
    purpose: purpose,
    remark: document.getElementById("remark").value,
    items: items
  };

  fetch(API_URL, {
    method: "POST",
    body: JSON.stringify(payload)
  })
  .then(res => res.json())
  .then(data => {
    if (data.status === "success") {
      alert("✅ Submitted successfully!");
    } else {
      alert("❌ " + data.message.join("\n"));
    }
  });
}
